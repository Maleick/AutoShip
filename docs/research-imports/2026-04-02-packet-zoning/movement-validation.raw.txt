  // Pseudocode for WickedSpeed packet modification
  void ModifyPositionPacket(uint8_t* packet_data, int opcode) {
      if (opcode != 0x1643) return;  // Only Position packets
      if (speed_a == speed_b) return; // No modification if speeds match
      if (!modification_enabled) return;

      // Read velocity field at packet offset 8 (callback offset 6)
      int32_t original = *(int32_t*)(packet_data + 8);

      // Sign-extend lower 10 bits to get the velocity value
      int32_t velocity = (original << 22) >> 22;  // Arithmetic shift

      // Calculate threshold
      float threshold = 0.7f * 40.0f - 70.0f;  // = -42.0

      // Compare velocity with threshold
      if ((float)velocity <= threshold) return;  // Already below threshold

      // Clamp to threshold value
      int32_t clamped = (int32_t)threshold;  // = -42

      // Apply clamped value while preserving lower 10 bits
      int32_t result = (original ^ clamped) & 0xFFFFFC00;
      result ^= clamped;

      // Write back
      *(int32_t*)(packet_data + 8) = result;
  }

  Bit Manipulation Explained

  The XOR/AND/XOR sequence replaces the upper 22 bits:
  original = AAAA AAAA AAAA AAAA AAAA AABB BBBB BBBB  (A=velocity, B=other)
  clamped  = CCCC CCCC CCCC CCCC CCCC CCXX XXXX XXXX  (C=new velocity, X=don't care)

  Step 1: XOR = (A XOR C) bits in upper, (B XOR X) in lower
  Step 2: AND 0xFFFFFC00 = keeps only upper 22 bits of XOR result, zeros lower 10
  Step 3: XOR with clamped = final value has C in upper 22, B in lower 10
